cmake_minimum_required(VERSION 3.24)
project(AntWorld VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(ANTWORLD_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

add_library(antworld_core src/types.cpp src/world_runtime.cpp)
target_include_directories(antworld_core PUBLIC include)
target_compile_features(antworld_core PUBLIC cxx_std_20)
set_target_properties(antworld_core PROPERTIES CXX_EXTENSIONS OFF)
add_library(antworld_scenario STATIC apps/scenario.cpp)
target_include_directories(antworld_scenario PUBLIC apps)
target_link_libraries(antworld_scenario PUBLIC antworld_core)
add_executable(antworld_headless apps/antworld_headless.cpp)
target_link_libraries(antworld_headless PRIVATE antworld_scenario)
add_executable(antworld_benchmark apps/antworld_benchmark.cpp)
target_link_libraries(antworld_benchmark PRIVATE antworld_core)

function(antworld_checks target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
        if(ANTWORLD_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    get_target_property(kind ${target} TYPE)
    if(MINGW AND kind STREQUAL "EXECUTABLE" AND NOT ANTWORLD_SANITIZERS)
        # Portable normal builds do not require compiler-runtime DLLs on PATH.
        target_link_options(${target} PRIVATE -static)
    endif()
endfunction()
foreach(target antworld_core antworld_scenario antworld_headless antworld_benchmark)
    antworld_checks(${target})
endforeach()
